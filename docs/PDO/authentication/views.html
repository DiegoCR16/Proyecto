 24def get_client_ip(request):
 25    """
 26    Obtiene la dirección IP del cliente desde la request HTTP.
 27    
 28    Args:
 29        request (HttpRequest): Objeto de petición HTTP de Django.
 30        
 31    Returns:
 32        str: Dirección IP del cliente.
 33    """
 34    x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
 35    if x_forwarded_for:
 36        ip = x_forwarded_for.split(',')[0]
 37    else:
 38        ip = request.META.get('REMOTE_ADDR')
 39    return ip

156    """
157    Callback del SSO de Keycloak tras autenticación exitosa.
158    Intercambia el código por tokens y obtiene roles y grupos directamente desde Keycloak (Fuente de Verdad Principal).
159    """

161    ip = get_client_ip(request)
162
163    if not code:
164        AuditLog.objects.create(
165            user=request.user if request.user.is_authenticated else None,
166            action="SSO_CALLBACK_FAILED",
167            ip_address=ip,
168            details="Código de autorización ausente en callback de Keycloak."
169        )
170        return HttpResponseBadRequest("Código de autorización ausente.")
171
172    token_url = f"{settings.KEYCLOAK_SERVER_URL}/realms/{settings.KEYCLOAK_REALM}/protocol/openid-connect/token"
173    token_data = {
174        'grant_type': 'authorization_code',
175        'client_id': settings.KEYCLOAK_CLIENT_ID,
176        'code': code,
177        'redirect_uri': settings.KEYCLOAK_REDIRECT_URI,
178    }
179    if getattr(settings, 'KEYCLOAK_CLIENT_SECRET', None):
180        token_data['client_secret'] = settings.KEYCLOAK_CLIENT_SECRET
181    
182    try:
183        token_response = requests.post(token_url, data=token_data)
184        if token_response.status_code != 200:
185            return HttpResponseBadRequest(f"Error al autenticar con Keycloak: {token_response.text}")
186        
187        token_json = token_response.json()
188        access_token = token_json.get('access_token')
189
190        token_parts = access_token.split('.')
191        payload_encoded = token_parts[1]
192        payload_encoded += '=' * (-len(payload_encoded) % 4)
193        payload_json = json.loads(base64.urlsafe_b64decode(payload_encoded).decode('utf-8'))
194
195        username = payload_json.get('preferred_username', 'keycloak_user')
196        email = payload_json.get('email', '')
197        kc_id = payload_json.get('sub')
198        
199        realm_roles = payload_json.get('realm_access', {}).get('roles', [])
200        client_roles = payload_json.get('resource_access', {}).get(settings.KEYCLOAK_CLIENT_ID, {}).get('roles', [])
201        all_roles = list(set(realm_roles + client_roles))
202
203        # Consultar Keycloak Admin API para obtener los grupos y roles exactos del usuario en Keycloak
204        kc_groups = []
205        try:
206            admin_token_url = f"{settings.KEYCLOAK_SERVER_URL}/realms/{settings.KEYCLOAK_REALM}/protocol/openid-connect/token"
207            admin_token_data = {
208                'grant_type': 'client_credentials',
209                'client_id': settings.KEYCLOAK_CLIENT_ID,
210                'client_secret': getattr(settings, 'KEYCLOAK_CLIENT_SECRET', ''),
211            }
212            admin_token_resp = requests.post(admin_token_url, data=admin_token_data, timeout=3)
213            if admin_token_resp.status_code == 200:
214                admin_token = admin_token_resp.json().get('access_token')
215                headers = {'Authorization': f'Bearer {admin_token}', 'Content-Type': 'application/json'}
216                
217                if not kc_id:
218                    search_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users?username={username}"
219                    search_resp = requests.get(search_url, headers=headers, timeout=3)
220                    if search_resp.status_code == 200 and search_resp.json():
221                        kc_id = search_resp.json()[0].get('id')
222
223                if kc_id:
224                    roles_api_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users/{kc_id}/role-mappings/realm"
225                    roles_api_resp = requests.get(roles_api_url, headers=headers, timeout=3)
226                    if roles_api_resp.status_code == 200:
227                        api_roles = [r.get('name') for r in roles_api_resp.json()]
228                        all_roles = list(set(all_roles + api_roles))
229
230                    groups_api_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users/{kc_id}/groups"
231                    groups_api_resp = requests.get(groups_api_url, headers=headers, timeout=3)
232                    if groups_api_resp.status_code == 200:
233                        kc_groups = groups_api_resp.json()
234        except Exception:
235            pass
236
237        all_roles_lower = [r.lower() for r in all_roles]
238
239        user, created = User.objects.get_or_create(username=username, defaults={'email': email})
240        if email and not user.email:
241            user.email = email
242            user.save()
243
244        role_obj = None
245        is_corp = False
246        
247        if any(r in all_roles_lower for r in ['admin', 'administrador', 'administrator']):
248            role_obj, _ = Role.objects.get_or_create(name="Admin")
249        elif any(r in all_roles_lower for r in ['cajero']):
250            role_obj, _ = Role.objects.get_or_create(name="Cajero")
251        elif any(r in all_roles_lower for r in ['analista']):
252            role_obj, _ = Role.objects.get_or_create(name="Analista")
253        elif any(r in all_roles_lower for r in ['corporate', 'corporativo', 'empresa', 'jefe']):
254            role_obj, _ = Role.objects.get_or_create(name="Corporate")
255            is_corp = True
256        else:
257            role_obj, _ = Role.objects.get_or_create(name="Cliente")
258
259        profile, _ = UserProfile.objects.get_or_create(user=user)
260        if kc_id:
261            profile.keycloak_id = kc_id
262        profile.role = role_obj
263        profile.is_corporate = is_corp
264        profile.save()
265
266        for g in kc_groups:
267            g_id = g.get('id')
268            g_name = g.get('name')
269            corp_group, _ = CorporateGroup.objects.get_or_create(
270                keycloak_group_id=g_id,
271                defaults={'group_name': g_name, 'juridica_profile': profile}
272            )
273            role_in_group = 'OPERADOR'
274            if any(r.lower() == f"{g_name.lower()}_cliente" or r.lower() == 'cliente' or r.lower() == f"{g_name.lower()}_jefe" or r.lower() == 'jefe' for r in all_roles_lower):
275                role_in_group = 'CLIENTE'
276            elif any(r.lower() == f"{g_name.lower()}_analista" or r.lower() == 'analista' for r in all_roles_lower):
277                role_in_group = 'ANALISTA'
278
279            GroupMembership.objects.get_or_create(
280                corporate_group=corp_group,
281                fisica_profile=profile,
282                defaults={'role_in_group': role_in_group}
283            )
284
285        login(request, user)
286        AuditLog.objects.create(
287            user=user,
288            action="SSO_LOGIN_SUCCESS",
289            ip_address=ip,
290            details=f"Inicio de sesión SSO exitoso. Roles Keycloak: {all_roles}, Grupos: {[g.get('name') for g in kc_groups]}"
291        )
292
293        if profile.requires_mfa() and not profile.itoken_verified:
294            return redirect('mfa_verify')
295        return redirect('dashboard_redirect')
296
297    except Exception as e:
298        return HttpResponseBadRequest(f"Error en comunicación con Keycloak: {str(e)}")
