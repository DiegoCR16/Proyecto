 19def get_client_ip(request):
 20    """
 21    Obtiene la dirección IP del cliente desde la request HTTP.
 22    
 23    Args:
 24        request (HttpRequest): Objeto de petición HTTP de Django.
 25        
 26    Returns:
 27        str: Dirección IP del cliente.
 28    """
 29    x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
 30    if x_forwarded_for:
 31        ip = x_forwarded_for.split(',')[0]
 32    else:
 33        ip = request.META.get('REMOTE_ADDR')
 34    return ip

 36@ensure_csrf_cookie
 37def login_view(request):
 38    """
 39    Vista de inicio de sesión principal. Muestra el acceso a Keycloak SSO
 40    y la pizarra pública de tasas de cambio en tiempo real (PSE-9) para invitados y usuarios.
 41    
 42    Args:
 43        request (HttpRequest): Objeto de petición HTTP de Django.
 44        
 45    Returns:
 46        HttpResponse: Respuesta renderizada con la plantilla de login y cotizaciones.
 47    """
 48    default_rates = [
 49        ('USD', 'Dólar Estadounidense', Decimal('7300.0000'), Decimal('7450.0000')),
 50        ('EUR', 'Euro', Decimal('7900.0000'), Decimal('8150.0000')),
 51        ('BRL', 'Real Brasileño', Decimal('1350.0000'), Decimal('1450.0000')),
 52        ('ARS', 'Peso Argentino', Decimal('7.5000'), Decimal('9.0000')),
 53        ('PYG', 'Guaraní Paraguayo', Decimal('1.0000'), Decimal('1.0000')),
 54    ]
 55
 56    for code, name, buy, sell in default_rates:
 57        ExchangeRate.objects.get_or_create(
 58            currency_code=code,
 59            defaults={
 60                'currency_name': name,
 61                'buy_rate': buy,
 62                'sell_rate': sell
 63            }
 64        )
 65
 66    rates = ExchangeRate.objects.all().order_by('id')
 67
 68    user_profile = None
 69    benefit_percentage = Decimal('0.00')
 70    benefit_label = 'Estándar'
 71    category_display = 'Invitado (Acceso Público)'
 72
 73    if request.user.is_authenticated:
 74        try:
 75            user_profile = request.user.profile
 76            category = user_profile.category
 77            if category == 'VIP':
 78                benefit_percentage = Decimal('2.00')
 79                benefit_label = '2% (VIP)'
 80                category_display = 'VIP'
 81            elif category == 'CORPORATIVO':
 82                benefit_percentage = Decimal('4.00')
 83                benefit_label = '4% (Corporativo)'
 84                category_display = 'Corporativo'
 85            else:
 86                category_display = 'Minorista'
 87        except Exception:
 88            pass
 89
 90    personalized_rates = []
 91    for rate in rates:
 92        if benefit_percentage > 0 and rate.currency_code != 'PYG':
 93            factor = Decimal('1.00') - (benefit_percentage / Decimal('100.00'))
 94            custom_sell = (rate.sell_rate * factor).quantize(Decimal('0.0001'))
 95            custom_buy = (rate.buy_rate / factor).quantize(Decimal('0.0001'))
 96        else:
 97            custom_sell = rate.sell_rate
 98            custom_buy = rate.buy_rate
 99
100        personalized_rates.append({
101            'currency_code': rate.currency_code,
102            'currency_name': rate.currency_name,
103            'standard_buy': rate.buy_rate,
104            'standard_sell': rate.sell_rate,
105            'custom_buy': custom_buy,
106            'custom_sell': custom_sell,
107            'last_updated': rate.last_updated,
108        })
109
110    context = {
111        'rates': personalized_rates,
112        'user_profile': user_profile,
113        'benefit_percentage': benefit_percentage,
114        'benefit_label': benefit_label,
115        'category_display': category_display,
116        'now': timezone.now(),
117    }
118
119    return render(request, 'authentication/login.html', context)

149@ensure_csrf_cookie
150def keycloak_callback_view(request):
151    """
152    Callback del SSO de Keycloak tras autenticación exitosa.
153    Intercambia el código por tokens y obtiene roles y grupos directamente desde Keycloak (Fuente de Verdad Principal).
154    """
155    code = request.GET.get('code')
156    ip = get_client_ip(request)
157
158    if not code:
159        AuditLog.objects.create(
160            user=request.user if request.user.is_authenticated else None,
161            action="SSO_CALLBACK_FAILED",
162            ip_address=ip,
163            details="Código de autorización ausente en callback de Keycloak."
164        )
165        return HttpResponseBadRequest("Código de autorización ausente.")
166
167    token_url = f"{settings.KEYCLOAK_SERVER_URL}/realms/{settings.KEYCLOAK_REALM}/protocol/openid-connect/token"
168    token_data = {
169        'grant_type': 'authorization_code',
170        'client_id': settings.KEYCLOAK_CLIENT_ID,
171        'code': code,
172        'redirect_uri': settings.KEYCLOAK_REDIRECT_URI,
173    }
174    if getattr(settings, 'KEYCLOAK_CLIENT_SECRET', None):
175        token_data['client_secret'] = settings.KEYCLOAK_CLIENT_SECRET
176    
177    try:
178        token_response = requests.post(token_url, data=token_data)
179        if token_response.status_code != 200:
180            return HttpResponseBadRequest(f"Error al autenticar con Keycloak: {token_response.text}")
181        
182        token_json = token_response.json()
183        access_token = token_json.get('access_token')
184
185        token_parts = access_token.split('.')
186        payload_encoded = token_parts[1]
187        payload_encoded += '=' * (-len(payload_encoded) % 4)
188        payload_json = json.loads(base64.urlsafe_b64decode(payload_encoded).decode('utf-8'))
189
190        username = payload_json.get('preferred_username', 'keycloak_user')
191        email = payload_json.get('email', '')
192        kc_id = payload_json.get('sub')
193        
194        realm_roles = payload_json.get('realm_access', {}).get('roles', [])
195        client_roles = payload_json.get('resource_access', {}).get(settings.KEYCLOAK_CLIENT_ID, {}).get('roles', [])
196        all_roles = list(set(realm_roles + client_roles))
197
198        # Consultar Keycloak Admin API para obtener los grupos y roles exactos del usuario en Keycloak
199        kc_groups = []
200        try:
201            admin_token_url = f"{settings.KEYCLOAK_SERVER_URL}/realms/{settings.KEYCLOAK_REALM}/protocol/openid-connect/token"
202            admin_token_data = {
203                'grant_type': 'client_credentials',
204                'client_id': settings.KEYCLOAK_CLIENT_ID,
205                'client_secret': getattr(settings, 'KEYCLOAK_CLIENT_SECRET', ''),
206            }
207            admin_token_resp = requests.post(admin_token_url, data=admin_token_data, timeout=3)
208            if admin_token_resp.status_code == 200:
209                admin_token = admin_token_resp.json().get('access_token')
210                headers = {'Authorization': f'Bearer {admin_token}', 'Content-Type': 'application/json'}
211                
212                if not kc_id:
213                    search_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users?username={username}"
214                    search_resp = requests.get(search_url, headers=headers, timeout=3)
215                    if search_resp.status_code == 200 and search_resp.json():
216                        kc_id = search_resp.json()[0].get('id')
217
218                if kc_id:
219                    roles_api_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users/{kc_id}/role-mappings/realm"
220                    roles_api_resp = requests.get(roles_api_url, headers=headers, timeout=3)
221                    if roles_api_resp.status_code == 200:
222                        api_roles = [r.get('name') for r in roles_api_resp.json()]
223                        all_roles = list(set(all_roles + api_roles))
224
225                    groups_api_url = f"{settings.KEYCLOAK_SERVER_URL}/admin/realms/{settings.KEYCLOAK_REALM}/users/{kc_id}/groups"
226                    groups_api_resp = requests.get(groups_api_url, headers=headers, timeout=3)
227                    if groups_api_resp.status_code == 200:
228                        kc_groups = groups_api_resp.json()
229        except Exception:
230            pass
231
232        all_roles_lower = [r.lower() for r in all_roles]
233
234        user, created = User.objects.get_or_create(username=username, defaults={'email': email})
235        if email and not user.email:
236            user.email = email
237            user.save()
238
239        role_obj = None
240        is_corp = False
241        
242        if any(r in all_roles_lower for r in ['admin', 'administrador', 'administrator']):
243            role_obj, _ = Role.objects.get_or_create(name="Admin")
244        elif any(r in all_roles_lower for r in ['cajero']):
245            role_obj, _ = Role.objects.get_or_create(name="Cajero")
246        elif any(r in all_roles_lower for r in ['analista']):
247            role_obj, _ = Role.objects.get_or_create(name="Analista")
248        elif any(r in all_roles_lower for r in ['corporate', 'corporativo', 'empresa', 'jefe']):
249            role_obj, _ = Role.objects.get_or_create(name="Corporate")
250            is_corp = True
251        else:
252            role_obj, _ = Role.objects.get_or_create(name="Cliente")
253
254        profile, _ = UserProfile.objects.get_or_create(user=user)
255        if kc_id:
256            profile.keycloak_id = kc_id
257        profile.role = role_obj
258        profile.is_corporate = is_corp
259        profile.save()
260
261        for g in kc_groups:
262            g_id = g.get('id')
263            g_name = g.get('name')
264            corp_group, _ = CorporateGroup.objects.get_or_create(
265                keycloak_group_id=g_id,
266                defaults={'group_name': g_name, 'juridica_profile': profile}
267            )
268            role_in_group = 'OPERADOR'
269            if any(r.lower() == f"{g_name.lower()}_cliente" or r.lower() == 'cliente' or r.lower() == f"{g_name.lower()}_jefe" or r.lower() == 'jefe' for r in all_roles_lower):
270                role_in_group = 'CLIENTE'
271            elif any(r.lower() == f"{g_name.lower()}_analista" or r.lower() == 'analista' for r in all_roles_lower):
272                role_in_group = 'ANALISTA'
273
274            GroupMembership.objects.get_or_create(
275                corporate_group=corp_group,
276                fisica_profile=profile,
277                defaults={'role_in_group': role_in_group}
278            )
279
280        login(request, user)
281        AuditLog.objects.create(
282            user=user,
283            action="SSO_LOGIN_SUCCESS",
284            ip_address=ip,
285            details=f"Inicio de sesión SSO exitoso. Roles Keycloak: {all_roles}, Grupos: {[g.get('name') for g in kc_groups]}"
286        )
287
288        if profile.requires_mfa() and not profile.itoken_verified:
289            return redirect('mfa_verify')
290        return redirect('dashboard_redirect')
291
292    except Exception as e:
293        return HttpResponseBadRequest(f"Error en comunicación con Keycloak: {str(e)}")

296def mfa_verify_view(request):
297    """
298    Vista para la verificación de iToken / MFA obligatorio para administrativos y corporativos.
299    
300    Args:
301        request (HttpRequest): Objeto de petición HTTP de Django.
302        
303    Returns:
304        HttpResponse: Renderiza la verificación MFA o redirige al panel.
305    """
306    ip = get_client_ip(request)
307    if not request.user.is_authenticated:
308        return redirect('login')
309
310    if request.method == 'POST':
311        itoken_code = request.POST.get('itoken_code')
312        if itoken_code == '123456':
313            profile, _ = UserProfile.objects.get_or_create(user=request.user)
314            profile.itoken_verified = True
315            profile.save()
316            AuditLog.objects.create(
317                user=request.user,
318                action="MFA_SUCCESS",
319                ip_address=ip,
320                details="Verificación de iToken/MFA exitosa."
321            )
322            return redirect('dashboard_redirect')
323        else:
324            AuditLog.objects.create(
325                user=request.user,
326                action="MFA_FAILED",
327                ip_address=ip,
328                details="Código iToken/MFA incorrecto."
329            )
330            return render(request, 'authentication/mfa_verify.html', {'error': 'iToken inválido. Use 123456 para pruebas.'})
331
332    return render(request, 'authentication/mfa_verify.html')

500def admin_roles_view(request):
501    """
502    Vista de administración para gestionar la descripción y asignación granular de permisos
503    de los roles fijos del sistema: Admin, Analista y Cliente (PSE-26).
504    
505    Args:
506        request (HttpRequest): Objeto de petición HTTP de Django.
507        
508    Returns:
509        HttpResponse: Renderiza la plantilla de gestión de roles y permisos con control RBAC.
510    """
511    profile, _ = UserProfile.objects.get_or_create(user=request.user)
512    role_name = profile.role.name.lower() if profile.role else ''
513    if not (request.user.is_superuser or 'admin' in role_name or 'administrador' in role_name):
514        AuditLog.objects.create(
515            user=request.user,
516            action="RBAC_ACCESS_DENIED",

522            'permissions': Permission.objects.none(),
523            'error': 'Acceso denegado: Se requiere rol de Administrador para gestionar roles y permisos.'
524        })
525
526    role_names = ['Admin', 'Analista', 'Cliente', 'Cajero', 'Jefe (Corporativo)', 'Operador (Corporativo)', 'Analista (Corporativo)']
527    descriptions = {
528        'Admin': 'Administrador General (Acceso Total - Protegido)',
529        'Analista': 'Analista de Operaciones (Permisos Delegados)',
530        'Cliente': 'Cliente de la Plataforma (Física y Jurídica General)',
531        'Cajero': 'Cajero de Sucursal (Operaciones de Caja)',
532        'Jefe (Corporativo)': 'Jefe / Propietario de Grupo Corporativo (Jurídica)',
533        'Operador (Corporativo)': 'Operador dentro del Grupo Corporativo (Jurídica)',
534        'Analista (Corporativo)': 'Analista dentro del Grupo Corporativo (Jurídica)',
535    }
536    for rname in role_names:
537        Role.objects.get_or_create(name=rname, defaults={'description': descriptions.get(rname, '')})

405            custom_sell = (rate.sell_rate * factor).quantize(Decimal('0.0001'))
406            custom_buy = (rate.buy_rate / factor).quantize(Decimal('0.0001'))
407        else:
408            custom_sell = rate.sell_rate
409            custom_buy = rate.buy_rate
410
411        personalized_rates.append({
412            'currency_code': rate.currency_code,
413            'currency_name': rate.currency_name,
414            'standard_buy': rate.buy_rate,
415            'standard_sell': rate.sell_rate,
416            'custom_buy': custom_buy,
417            'custom_sell': custom_sell,
418            'last_updated': rate.last_updated,
419        })
420
421    user_mode = request.session.get('user_mode', False)
422
423    context = {
424        'profile': profile,
425        'rates': personalized_rates,
426        'benefit_percentage': benefit_percentage,
427        'benefit_label': benefit_label,
428        'category_display': category_display,
429        'now': timezone.now(),
430        **interface_ctx,
431    }
432
433    if interface_ctx['is_admin']:
434        return render(request, 'authentication/admin_dashboard.html', context)
435    elif profile.role and 'cajero' in profile.role.name.lower():
436        return render(request, 'authentication/cajero_dashboard.html', context)
437    elif profile.role and 'analista' in profile.role.name.lower():
438        return render(request, 'authentication/analista_dashboard.html', context)
439    elif profile.is_corporate or 'corporativo' in role_name:
440        if user_mode:
441            return render(request, 'authentication/user_dashboard.html', context)
442        return render(request, 'authentication/corporate_dashboard.html', context)
443    elif has_client_mode:
444        if user_mode:
445            return render(request, 'authentication/user_dashboard.html', context)
446        return render(request, 'authentication/client_dashboard.html', context)
447    else:
448        return render(request, 'authentication/user_dashboard.html', context)

501def admin_roles_view(request):
502    """
503    Vista de administración para gestionar la descripción y asignación granular de permisos

505    
506    Args:
507        request (HttpRequest): Objeto de petición HTTP de Django.

509    Returns:
510        HttpResponse: Renderiza la plantilla de gestión de roles y permisos con control RBAC.
511    """
512    profile, _ = UserProfile.objects.get_or_create(user=request.user)
513    role_name = profile.role.name.lower() if profile.role else ''
514    if not (request.user.is_superuser or 'admin' in role_name or 'administrador' in role_name):
515        AuditLog.objects.create(
516            user=request.user,
517            action="RBAC_ACCESS_DENIED",
518            ip_address=get_client_ip(request),
519            details="Intento de acceso no autorizado al módulo de gestión de roles (PSE-26)."

522            'roles': Role.objects.none(),
523            'permissions': Permission.objects.none(),
524            'error': 'Acceso denegado: Se requiere rol de Administrador para gestionar roles y permisos.'
525        })
526
527    role_names = ['Admin', 'Analista', 'Cliente', 'Cajero', 'Jefe (Corporativo)', 'Operador (Corporativo)', 'Analista (Corporativo)']
528    descriptions = {
529        'Admin': 'Administrador General (Acceso Total - Protegido)',
530        'Analista': 'Analista de Operaciones (Permisos Delegados)',
531        'Cliente': 'Cliente de la Plataforma (Física y Jurídica General)',
532        'Cajero': 'Cajero de Sucursal (Operaciones de Caja)',
533        'Jefe (Corporativo)': 'Jefe / Propietario de Grupo Corporativo (Jurídica)',
534        'Operador (Corporativo)': 'Operador dentro del Grupo Corporativo (Jurídica)',
535        'Analista (Corporativo)': 'Analista dentro del Grupo Corporativo (Jurídica)',
536    }
537    for rname in role_names:
538        Role.objects.get_or_create(name=rname, defaults={'description': descriptions.get(rname, '')})
539
540    default_perms = [
541        ('can_manage_roles', 'Gestionar Roles y Permisos', 'Crear, editar, desactivar roles y asignar permisos (Admin/Analista)'),
542        ('can_manage_clients', 'Gestionar Clientes', 'Crear, editar y asociar cuentas de clientes y grupos corporativos (Admin/Analista)'),
543        ('can_manage_rates', 'Gestionar Tasas de Cambio', 'Actualizar cotizaciones y tasas de cambio en tiempo real (Admin/Analista)'),
544        ('can_view_audit', 'Ver Auditoría', 'Consultar logs de auditoría e incidentes de seguridad (Admin/Analista)'),
545        ('can_perform_exchange', 'Realizar Transacciones', 'Ejecutar compra y venta de divisas en la interfaz de cliente'),
546        ('can_view_rates', 'Ver Pizarra de Tasas', 'Consultar cotizaciones actualizadas en tiempo real'),
547    ]
548    for codename, name, desc in default_perms:
549        Permission.objects.get_or_create(codename=codename, defaults={'name': name, 'description': desc})
550
551    ip = get_client_ip(request)
552
553    if request.method == 'POST':
554        action = request.POST.get('action')
555        
556        if action == 'edit':
557            role_id = request.POST.get('role_id')
558            role = get_object_or_404(Role, id=role_id)
559            
560            if role.name.lower() in ['admin', 'administrador']:
561                return render(request, 'authentication/admin_roles.html', {
562                    'roles': Role.objects.filter(name__in=role_names).prefetch_related('permissions'),
563                    'permissions': Permission.objects.all(),

565                })

568            permission_ids = request.POST.getlist('permissions')
569
570            if role.name.lower() == 'cliente':

572                selected_perms = Permission.objects.filter(id__in=permission_ids)
573                if selected_perms.filter(codename__in=admin_codenames).exists():
574                    return render(request, 'authentication/admin_roles.html', {
575                        'roles': Role.objects.filter(name__in=role_names).prefetch_related('permissions'),
576                        'permissions': Permission.objects.all(),

578                    })

580            if role.name.lower() == 'analista':
581                selected_perms = Permission.objects.filter(id__in=permission_ids)
582                if selected_perms.filter(codename='can_perform_exchange').exists():
583                    return render(request, 'authentication/admin_roles.html', {
584                        'roles': Role.objects.filter(name__in=role_names).prefetch_related('permissions'),
585                        'permissions': Permission.objects.all(),
586                        'error': 'Error de validación: El rol Analista no realiza transacciones de cliente.'
587                    })

591            role.permissions.set(Permission.objects.filter(id__in=permission_ids))

593            try:

604                    headers = {'Authorization': f'Bearer {admin_token}', 'Content-Type': 'application/json'}

606            except Exception as e:

610                user=request.user,
611                action="ROLE_UPDATE",
612                ip_address=ip,
613                details=f"Modificación del rol fijo {role.name}."
614            )

623        'now': timezone.now(),
