 16        Obtiene la URL base del servicio de Factura Segura desde la configuración de Django (settings).

 20    @staticmethod
 21    def _respuesta_error(response):
 22        """
 23        Convierte una respuesta HTTP fallida en el formato de error del servicio.
 24
 25        Args:
 26            response: Respuesta HTTP distinta de 200.
 27
 28        Returns:
 29            dict: Código HTTP, descripción legible y cuerpo original de la respuesta.
 30        """
 31        try:
 32            errores = response.json().get('response', {}).get('errors') or []
 33        except (ValueError, AttributeError):
 34            errores = []
 35        descripcion = ', '.join(str(e) for e in errores) or f"HTTP {response.status_code}"
 36        return {"code": response.status_code, "description": descripcion, "error": response.text}

 38    @classmethod
 39    def obtener_token(cls, email=None, password=None):
 40        """
 41        Se conecta al endpoint de login para obtener el authentication_token.
 42        Utiliza credenciales de settings o variables de entorno si no se proveen explícitamente.
 43        
 44        Args:
 45            email (str): Correo electrónico de acceso a Factura Segura.
 46            password (str): Contraseña de acceso.
 47            
 48        Returns:
 49            str: Token de autenticación si es exitoso, None en caso contrario.
 50        """
 51        email = email or getattr(settings, 'SIFEN_API_EMAIL', 'test@facturasegura.com.py')
 52        password = password or getattr(settings, 'SIFEN_API_PASSWORD', 'testpassword')
 53        
 54        base_url = cls._get_base_url()
 55        login_url = f"{base_url}/login?include_auth_token"
 56        
 57        headers = {
 58            'accept': 'application/json',
 59            'Content-Type': 'application/json'
 60        }
 61        payload = {
 62            'email': email,
 63            'password': password
 64        }
 65        
 66        try:
 67            response = requests.post(login_url, json=payload, headers=headers, timeout=30)
 68            data = response.json()
 69            if response.status_code == 200:
 70                resp_dict = data.get('response', {})
 71                token = (
 72                    data.get('authentication_token') or 
 73                    data.get('auth_token') or 
 74                    data.get('token') or 
 75                    resp_dict.get('authentication_token') or
 76                    resp_dict.get('user', {}).get('authentication_token')
 77                )
 78                if not token:
 79                    logger.error("Login SIFEN sin authentication_token en la respuesta")
 80                return token
 81            logger.error(f"Error en autenticación SIFEN: {response.status_code} - {response.text}")
 82            return None
 83        except requests.exceptions.RequestException as e:
 84            logger.error(f"Excepción de red al autenticar con SIFEN: {str(e)}")
 85            return None

 88    def generar_de(cls, auth_token, transaccion_data):
 89        """
 90        Construye el payload JSON exigido por Factura Segura y envía la operación 'generar_de'
 91        para procesar la transacción de compra/venta de divisas.
 92        
 93        Args:
 94            auth_token (str): Token de autenticación activo.
 95            transaccion_data (dict): Datos de la transacción de divisas completada.
 96            
 97        Returns:
 98            dict: Respuesta con el CDC generado si es exitoso (code: 0) o detalles del error.
 99        """
100        base_url = cls._get_base_url()
101        esi_url = f"{base_url}/misife00/v1/esi"
102        
103        headers = {
104            'accept': 'application/json',
105            'Content-Type': 'application/json',
106            'Authentication-Token': auth_token
107        }
108        
109        default_emis = {
110            "dRucEm": "2595733",
111            "dDVEmi": "3",
112            "iTipCont": "1",
113            "dNumTim": "02595733",
114            "dFeIniT": "2025-03-27",
115            "dFelIniT": "2025-03-27",
116            "dEst": "001",
117            "dPunExp": "003",
118            "dTelEmi": "(021)5887000"
119        }
120        user_emis = transaccion_data.get("emis", {})
121        merged_emis = {**default_emis, **user_emis}
122
123        de_data = transaccion_data.get("DE", {
124            "dVerFor": 150,
125            "gEmis": merged_emis,
126            "gDatGralOpe": transaccion_data.get("dat_gral_ope", {}),
127            "gDtipDE": transaccion_data.get("dtip_de", {})
128        })
129        if "gEmis" in de_data and isinstance(de_data["gEmis"], dict):
130            de_data["gEmis"] = {**default_emis, **de_data["gEmis"]}
131
132        if isinstance(de_data, dict) and "dEmailRec" not in de_data:
133            de_data["dEmailRec"] = transaccion_data.get("dEmailRec", "soporte@facturasegura.com.py")
134
135        if isinstance(de_data, dict) and "gActEco" not in de_data:
136            de_data["gActEco"] = getattr(settings, 'FACTURASEGURA_EMISOR_ACTIVIDADES_ECONOMICAS', [
137                {
138                    'cActEco': '62010',
139                    'dDesActEco': 'Actividades de programación informática'
140                },
141                {
142                    'cActEco': '74909',
143                    'dDesActEco': 'Otras actividades profesionales, científicas y técnicas n.c.p.'
144                }
145            ])
146
147
148
149        payload = {
150            "operation": "generar_de",
151            "params": {
152                "DE": de_data
153            }
154        }
155        
156        try:
157            logger.debug("Payload enviado a SIFEN: %s", payload)
158            response = requests.post(esi_url, json=payload, headers=headers, timeout=30)
159            if response.status_code == 200:
160                return response.json()
161            logger.error(f"Error al generar DE en SIFEN: {response.status_code} - {response.text}")
162            return cls._respuesta_error(response)
163        except requests.exceptions.RequestException as e:
164            logger.error(f"Excepción de red al generar DE en SIFEN: {str(e)}")
165            return {"code": -1, "error": str(e)}
