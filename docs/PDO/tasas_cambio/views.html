131def get_user_effective_category(user, request=None):
132    """
133    Determina la categoría efectiva del cliente con el que se está operando,
134    considerando si el usuario está en 'modo usuario' (user_mode=True), en cuyo caso
135    opera como usuario regular sin beneficios de cliente.
136    Si está en modo cliente, considera el cliente activo en sesión (active_client_id),
137    las relaciones en UsuarioClienteRelacion, o el perfil del usuario.
138    """
139    cat_code = 'MINORISTA'
140    if not user or not user.is_authenticated:
141        return cat_code
142
143    try:
144        if request and request.session.get('user_mode', False):
145            return 'MINORISTA'
146
147        active_client = None
148        active_client_id = request.session.get('active_client_id') if request else None
149
150        if active_client_id:
151            active_client = Cliente.objects.filter(id=active_client_id).first()
152            if not active_client and str(active_client_id).isdigit():
153                active_client = Cliente.objects.filter(id=int(active_client_id)).first()
154
155        if not active_client and hasattr(user, 'profile') and user.profile.keycloak_id:
156            rel = UsuarioClienteRelacion.objects.filter(keycloak_user_id=user.profile.keycloak_id).select_related('cliente').first()
157            if rel:
158                active_client = rel.cliente
159                if request:
160                    request.session['active_client_id'] = str(active_client.id)
161
162        if active_client:
163            cat_code = active_client.categoria
164        elif hasattr(user, 'profile') and user.profile.category:
165            cat_code = user.profile.category
166    except Exception:
167        pass
168
169    return cat_code

172class SimuladorConversionService:
173    """
174    Servicio de dominio para la simulación de conversión de divisas (PSE-11 / PSE-29).
175    Calcula conversiones exactas aplicando tipos de cambio vigentes y beneficios dinámicos por categoría y umbrales.
176    """
177
178    @staticmethod
179    def simular(from_currency, to_currency, amount, user=None, request=None):
180        """
181        Simula una conversión monetaria entre dos divisas aplicando reglas de beneficio parametrizables.
182
183        Args:
184            from_currency (str): Código ISO de la moneda de origen (ej. 'USD', 'PYG').
185            to_currency (str): Código ISO de la moneda de destino (ej. 'PYG', 'EUR').
186            amount (Decimal or float or int): Monto a convertir.
187            user (User, optional): Usuario solicitante para determinar categoría y beneficios (VIP, Corporativo).
188
189        Returns:
190            dict: Diccionario con el resultado detallado de la simulación y desglose transparente.
191
192        Raises:
193            ValidationError: Si faltan campos obligatorios, el monto es inválido o no existe la tasa de cambio.
194        """
195        if not from_currency or not to_currency or amount is None:
196            raise ValidationError("Los campos moneda origen, moneda destino y monto son obligatorios.")
197
198        try:
199            amount_dec = Decimal(str(amount))
200        except (ValueError, TypeError):
201            raise ValidationError("El monto ingresado debe ser un valor numérico válido.")
202
203        if amount_dec <= Decimal('0.00'):
204            raise ValidationError("El monto de la simulación debe ser mayor a cero.")
205
206        ensure_default_benefit_rules()
207        benefit_percentage = Decimal('0.00')
208        category_name = 'Invitado / Minorista'
209        threshold_met = False
210        min_operation_amount = Decimal('0.00')
211
212        if user and user.is_authenticated:
213            try:
214                cat_code = get_user_effective_category(user, request=request)
215                rule = ClientBenefitRule.objects.filter(category_code=cat_code).first()
216                if rule:
217                    category_name = rule.category_name
218                    min_operation_amount = rule.min_operation_amount
219
220                    # Evaluar equivalencia en PYG para umbral transaccional
221                    eval_amount = amount_dec
222                    if from_currency != 'PYG':
223                        try:
224                            r_from = ExchangeRate.objects.get(currency_code=from_currency)
225                            eval_amount = amount_dec * r_from.buy_rate
226                        except Exception:
227                            pass
228
229                    if eval_amount >= min_operation_amount:
230                        benefit_percentage = rule.benefit_percentage
231                        threshold_met = True
232                    else:
233                        benefit_percentage = Decimal('0.00')
234                        threshold_met = False
235            except Exception:
236                pass
237
238        factor = Decimal('1.00') - (benefit_percentage / Decimal('100.00'))
239        transparent_breakdown = f"Categoría: {category_name} | Umbral Mínimo: ₲ {min_operation_amount:,.2f} | Beneficio: {benefit_percentage}% {'(Aplicado)' if threshold_met else '(No alcanza umbral)'}"
240
241        if from_currency == to_currency:
242            return {
243                'from_currency': from_currency,
244                'to_currency': to_currency,
245                'amount': amount_dec,
246                'converted_amount': amount_dec,
247                'applied_rate': Decimal('1.0000'),
248                'standard_rate': Decimal('1.0000'),
249                'operation_type': 'IDÉNTICA',
250                'benefit_percentage': benefit_percentage,
251                'category_name': category_name,
252                'threshold_met': threshold_met,
253                'min_operation_amount': min_operation_amount,
254                'transparent_breakdown': transparent_breakdown,
255            }
256
257        if from_currency == 'PYG':
258            try:
259                rate_obj = ExchangeRate.objects.get(currency_code=to_currency)
260            except ExchangeRate.DoesNotExist:
261                raise ValidationError(f"El sistema no cuenta con la tasa de cambio registrada para la divisa: {to_currency}")
262
263            standard_rate = rate_obj.sell_rate
264            if standard_rate <= 0:
265                raise ValidationError(f"La tasa de venta para {to_currency} no se encuentra disponible.")
266
267            custom_rate = (standard_rate * factor).quantize(Decimal('0.0001'))
268            converted_amount = (amount_dec / custom_rate).quantize(Decimal('0.01'))
269
270            return {
271                'from_currency': from_currency,
272                'to_currency': to_currency,
273                'amount': amount_dec,
274                'converted_amount': converted_amount,
275                'applied_rate': custom_rate,
276                'standard_rate': standard_rate,
277                'operation_type': 'COMPRA',
278                'benefit_percentage': benefit_percentage,
279                'category_name': category_name,
280                'threshold_met': threshold_met,
281                'min_operation_amount': min_operation_amount,
282                'transparent_breakdown': transparent_breakdown,
283            }
284
285        elif to_currency == 'PYG':
286            try:
287                rate_obj = ExchangeRate.objects.get(currency_code=from_currency)
288            except ExchangeRate.DoesNotExist:
289                raise ValidationError(f"El sistema no cuenta con la tasa de cambio registrada para la divisa: {from_currency}")
290
291            standard_rate = rate_obj.buy_rate
292            if standard_rate <= 0:
293                raise ValidationError(f"La tasa de compra para {from_currency} no se encuentra disponible.")
294
295            if benefit_percentage > 0:
296                custom_rate = (standard_rate / factor).quantize(Decimal('0.0001'))
297            else:
298                custom_rate = standard_rate
299
300            converted_amount = (amount_dec * custom_rate).quantize(Decimal('0.01'))
301
302            return {
303                'from_currency': from_currency,
304                'to_currency': to_currency,
305                'amount': amount_dec,
306                'converted_amount': converted_amount,
307                'applied_rate': custom_rate,
308                'standard_rate': standard_rate,
309                'operation_type': 'VENTA',
310                'benefit_percentage': benefit_percentage,
311                'category_name': category_name,
312                'threshold_met': threshold_met,
313                'min_operation_amount': min_operation_amount,
314                'transparent_breakdown': transparent_breakdown,
315            }
316
317        else:
318            try:
319                rate_from = ExchangeRate.objects.get(currency_code=from_currency)
320            except ExchangeRate.DoesNotExist:
321                raise ValidationError(f"El sistema no cuenta con la tasa de cambio registrada para la divisa: {from_currency}")
322
323            try:
324                rate_to = ExchangeRate.objects.get(currency_code=to_currency)
325            except ExchangeRate.DoesNotExist:
326                raise ValidationError(f"El sistema no cuenta con la tasa de cambio registrada para la divisa: {to_currency}")
327
328            std_buy = rate_from.buy_rate
329            std_sell = rate_to.sell_rate
330            if std_buy <= 0 or std_sell <= 0:
331                raise ValidationError("Tasas de cambio inválidas para la conversión cruzada.")
332
333            if benefit_percentage > 0:
334                custom_buy = (std_buy / factor).quantize(Decimal('0.0001'))
335                custom_sell = (std_sell * factor).quantize(Decimal('0.0001'))
336            else:
337                custom_buy = std_buy
338                custom_sell = std_sell
339
340            pyg_amount = amount_dec * custom_buy
341            converted_amount = (pyg_amount / custom_sell).quantize(Decimal('0.01'))
342            effective_rate = (custom_buy / custom_sell).quantize(Decimal('0.0004'))
343
344            return {
345                'from_currency': from_currency,
346                'to_currency': to_currency,
347                'amount': amount_dec,
348                'converted_amount': converted_amount,
349                'applied_rate': effective_rate,
350                'standard_rate': std_sell,
351                'operation_type': 'CRUZADA',
352                'benefit_percentage': benefit_percentage,
353                'category_name': category_name,
354                'threshold_met': threshold_met,
355                'min_operation_amount': min_operation_amount,
356                'transparent_breakdown': transparent_breakdown,
357            }

476def populate_mock_history():
477    """
478    Pobla datos históricos simulados de tasas de cambio si la tabla ExchangeRateHistory está vacía.
479    Genera registros diarios para el último año para las principales divisas (USD, EUR, BRL, ARS).
480    """
481    if ExchangeRateHistory.objects.exists():
482        return
483
484    base_rates = {
485        'USD': (Decimal('7300.0000'), Decimal('7450.0000')),
486        'EUR': (Decimal('7900.0000'), Decimal('8150.0000')),
487        'BRL': (Decimal('1350.0000'), Decimal('1450.0000')),
488        'ARS': (Decimal('7.5000'), Decimal('9.0000')),
489        'PYG': (Decimal('1.0000'), Decimal('1.0000')),
490    }
491
492    now = timezone.now()
493    random.seed(42)  # Semilla fija para consistencia en pruebas
494    for code, (base_buy, base_sell) in base_rates.items():
495        current_buy = base_buy
496        current_sell = base_sell
497        for i in range(365, -1, -1):
498            day_time = now - timedelta(days=i)
499            if code != 'PYG':
500                var_buy = float(current_buy) * random.uniform(-0.005, 0.005)
501                var_sell = float(current_sell) * random.uniform(-0.005, 0.005)
502                current_buy = (current_buy + Decimal(str(var_buy))).quantize(Decimal('0.0001'))
503                current_sell = (current_sell + Decimal(str(var_sell))).quantize(Decimal('0.0001'))
504
505            ExchangeRateHistory.objects.create(
506                currency_code=code,
507                buy_rate=current_buy,
508                sell_rate=current_sell,
509                timestamp=day_time
510            )

513def rates_evolution_api(request):
514    """
515    API JSON para obtener el historial de evolución de tasas de cambio (PSE-10).
516    
517    Args:
518        request (HttpRequest): Solicitud GET con parámetros 'currency' (ej. USD) y 'range' (1D, 7D, 1M, 1Y).
519        
520    Returns:
521        JsonResponse: Datos estructurados con etiquetas, tasas de compra, venta, estado y mensajes.
522    """
523    populate_mock_history()
524
525    currency = request.GET.get('currency', 'USD').upper()
526    time_range = request.GET.get('range', '30D').upper()
527
528    now = timezone.now()
529    
530    if time_range == '1D':
531        start_date = now - timedelta(days=1)
532    elif time_range == '7D':
533        start_date = now - timedelta(days=7)
534    elif time_range == '1M' or time_range == '30D':
535        start_date = now - timedelta(days=30)
536    elif time_range == '1Y':
537        start_date = now - timedelta(days=365)
538    else:
539        start_date = now - timedelta(days=30)
540
541    history_qs = ExchangeRateHistory.objects.filter(
542        currency_code=currency,
543        timestamp__gte=start_date
544    ).order_by('timestamp')
545
546    if not history_qs.exists():
547        return JsonResponse({
548            'has_data': False,
549            'currency': currency,
550            'range': time_range,
551            'message': f'No se registran datos históricos suficientes para la divisa {currency} en el rango seleccionado ({time_range}).',
552            'labels': [],
553            'buy_rates': [],
554            'sell_rates': [],
555        })
556
557    labels = []
558    buy_rates = []
559    sell_rates = []
560
561    for record in history_qs:
562        if time_range == '1D':
563            label = record.timestamp.strftime('%H:%M')
564        elif time_range in ['7D', '1M', '30D']:
565            label = record.timestamp.strftime('%d/%m/%Y')
566        else:
567            label = record.timestamp.strftime('%m/%Y')
568
569        labels.append(label)
570        buy_rates.append(float(record.buy_rate))
571        sell_rates.append(float(record.sell_rate))
572
573    return JsonResponse({
574        'has_data': True,
575        'currency': currency,
576        'range': time_range,
577        'message': '',
578        'labels': labels,
579        'buy_rates': buy_rates,
580        'sell_rates': sell_rates,
581    })
