 88        """Estado de la factura electrónica (Emitida, Aprobada, Rechazada, Pendiente, En cola, Error)."""
 89        if self.transparent_breakdown:
 90            if "SIFEN Estado: Error" in self.transparent_breakdown:
 91                return "Error"
 92            if "SIFEN Estado: En cola" in self.transparent_breakdown and "SIFEN CDC:" not in self.transparent_breakdown:
 93                return "En cola"
 94            if "SIFEN Estado: Rechazada" in self.transparent_breakdown:
 95                return "Rechazada"
 96            if "SIFEN Estado: Aprobada" in self.transparent_breakdown:
 97                return "Aprobada"
 98            if "SIFEN Estado: Pendiente" in self.transparent_breakdown:
 99                return "Pendiente"
100            if "SIFEN Estado: Emitida" in self.transparent_breakdown:
101                return "Emitida"
102        return "Emitida"

119            update_fields = kwargs.get('update_fields')
120            if update_fields and list(update_fields) == ['transparent_breakdown']:
121                pass
122            elif old_obj.status == 'PENDING' and self.status in ['SUCCESS', 'CANCELLED', 'FAILED']:
123                pass
